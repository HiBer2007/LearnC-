**一个类加上 `virtual`，对象就不再是它看上去的样子。** 字段还是那些字段，
但对象开头多出一个指针，指向一张编译器生成、放在只读段里的表；
`sizeof` 会跳一档，两个编译器可能给出不同的数，跨语言传递时那多出来的几个字节会顶掉约定好的布局。

《05-类与面向对象/08-多态：重载、虚函数与它们的分工.md》第 3.2 小节已经把这笔账算过一遍：
`int` 从 4 字节变成 16 字节，一个虚函数与两个虚函数一样大，多继承时每个多态基类子对象各带一个指针。
那一章停在「对象里多了一个指针」这一层，因为语言层面能说的就到这里。
《05-类与面向对象/07-继承.md》第 6.4 小节讲虚继承时，也只讲到「共同的基类只留一份」这个语义，
布局怎么实现留给了本章。

**本章把那张表打开来看。** 编译器把哪些地址写进了表、表的第几个槽位放什么、
`type_info` 指针为什么在 `vptr[-1]`、多继承时 `this` 指针被加了多少、
虚基类的偏移存在哪里、`dynamic_cast` 一次要跑多少纳秒——这些都能在本机跑出来、量出数字。
最后把同一份代码交给 `arm-none-eabi-g++` 编成 Cortex-M 的固件，
用 `size` 与 `objdump -h` 测一次 flash 与 RAM 的占用，看这些机制在一块几十 KB 的单片机上要付多少钱。

这些结论**不是标准规定的**。标准只保证语义，二进制长什么样由 ABI 定；
本章给出的每个偏移、每个槽位，都来自本机工具链的真实输出，
换编译器、换版本、换平台都可能不同。

# 第 1 节 对象里多了什么

## 1.1 从 `sizeof` 阶梯开始

`sizeof` 是与实现有关的量，两台机器的结果可以直接摆在一起比。
同一个头文件里的八个类，在 x86-64 与 Cortex-M 上各量一遍：

`C++`

```cpp
/* ladder.cpp    编译：g++ -std=c++17 ladder.cpp -o ladder */
#include <cstdio>
#include <cstddef>

/* 第一组：虚函数带来的台阶 */
class NoVirtual   { public: int a = 1; };
class OneVirtual  { public: int a = 1; virtual void f() { } };
class TwoVirtual  { public: int a = 1; virtual void f() { } virtual void g() { } };
class FourVirtual { public: int a = 1;
                    virtual void f() { } virtual void g() { }
                    virtual void h() { } virtual void k() { } };

/* 第二组：空类与空基类 */
class Empty { };
class EmptyDerived : public Empty { };
class EmptyMember { Empty e; int a = 1; };

/* 第三组：多重继承 */
class Base1 { public: virtual void f() { } };
class Base2 { public: virtual void g() { } };
class MI    : public Base1, public Base2 { public: int m = 3; };

/* 第四组：虚继承与菱形 */
class VA { public: int a = 1; virtual void fa() { } };
class VB : virtual public VA { public: int b = 2; };
class VC : virtual public VA { public: int c = 3; };
class VD : public VB, public VC { public: int d = 4; };

/* 帮手：成员相对对象开头的偏移（多态类型不是标准布局，offsetof 不适用） */
template <class T, class M>
static long off(const T &obj, const M &member) {
    return reinterpret_cast<const char *>(&member) - reinterpret_cast<const char *>(&obj);
}

int main() {
    std::printf("=== 第一组：虚函数的台阶\n");
    std::printf("NoVirtual   = %zu\n", sizeof(NoVirtual));
    std::printf("OneVirtual  = %zu\n", sizeof(OneVirtual));
    std::printf("TwoVirtual  = %zu\n", sizeof(TwoVirtual));
    std::printf("FourVirtual = %zu\n", sizeof(FourVirtual));

    std::printf("=== 第二组：空类\n");
    std::printf("Empty       = %zu\n", sizeof(Empty));
    std::printf("EmptyDerived= %zu\n", sizeof(EmptyDerived));
    std::printf("EmptyMember = %zu\n", sizeof(EmptyMember));

    std::printf("=== 第三组：多重继承\n");
    std::printf("Base1 = %zu, Base2 = %zu, MI = %zu\n",
                sizeof(Base1), sizeof(Base2), sizeof(MI));
    MI mi;
    std::printf("MI 对象开头 %p\n", static_cast<const void *>(&mi));
    std::printf("  Base1 子对象 %p 相对偏移 %ld\n",
                static_cast<const void *>(static_cast<Base1 *>(&mi)),
                off(mi, *static_cast<Base1 *>(&mi)));
    std::printf("  Base2 子对象 %p 相对偏移 %ld\n",
                static_cast<const void *>(static_cast<Base2 *>(&mi)),
                off(mi, *static_cast<Base2 *>(&mi)));
    std::printf("  成员 m      %p 相对偏移 %ld\n",
                static_cast<const void *>(&mi.m), off(mi, mi.m));

    std::printf("=== 第四组：虚继承\n");
    std::printf("VA=%zu VB=%zu VC=%zu VD=%zu\n",
                sizeof(VA), sizeof(VB), sizeof(VC), sizeof(VD));
    VD vd;
    std::printf("VD 对象开头 %p\n", static_cast<const void *>(&vd));
    std::printf("  VB 子对象偏移 %ld\n", off(vd, *static_cast<VB *>(&vd)));
    std::printf("  VC 子对象偏移 %ld\n", off(vd, *static_cast<VC *>(&vd)));
    std::printf("  VA 子对象偏移 %ld\n", off(vd, *static_cast<VA *>(&vd)));
    std::printf("  d 偏移 %ld, b 偏移 %ld, c 偏移 %ld, a 偏移 %ld\n",
                off(vd, vd.d), off(vd, vd.b), off(vd, vd.c), off(vd, vd.a));
    return 0;
}
```

```cpp
   qemu-system-arm -M netduinoplus2 -kernel mcu_ladder.elf -nographic \
     -semihosting-config enable=on,target=native */

class OneVirtual  { public: int a = 1; virtual void f() { } };
class TwoVirtual  { public: int a = 1; virtual void f() { } virtual void g() { } };

class Empty { };
class EmptyDerived : public Empty { };
class EmptyMember { Empty e; int a = 1; };

class Base1 { public: virtual void f() { } };
class Base2 { public: virtual void g() { } };
class MI    : public Base1, public Base2 { public: int m = 3; };

class VB : virtual public VA { public: int b = 2; };
class VC : virtual public VA { public: int c = 3; };
class VD : public VB, public VC { public: int d = 4; };

MI g_mi;
VD g_vd;

/* 两个对象的相对距离：用真实对象取地址，不用空指针 */

    return (uint32_t)((const char *)member - (const char *)base);
}

    sh_row("TwoVirtual  = ", sizeof(TwoVirtual));

    sh_row("EmptyMember = ", sizeof(EmptyMember));
    sh_row("Base1 = ", sizeof(Base1));
    sh_row("Base2 = ", sizeof(Base2));
    sh_row("MI    = ", sizeof(MI));

    sh_row("VB = ", sizeof(VB));
    sh_row("VC = ", sizeof(VC));
    sh_row("VD = ", sizeof(VD));
    sh_row("VD 里 VB 子对象偏移  = ", rel(&g_vd, static_cast<VB *>(&g_vd)));
    sh_row("VD 里 VC 子对象偏移  = ", rel(&g_vd, static_cast<VC *>(&g_vd)));
    sh_row("VD 里 VA 子对象偏移  = ", rel(&g_vd, static_cast<VA *>(&g_vd)));
    sh_row("VD 里 d 的偏移       = ", rel(&g_vd, &g_vd.d));
    sh_row("VD 里 a 的偏移       = ", rel(&g_vd, &g_vd.a));
    for (;;) { }
}
```

四组数字合起来读，规则是清楚的：

| 现象 | 原因 |
|---|---|
| `NoVirtual` 4 字节，`OneVirtual` 16 字节 | 对象开头多了一个 8 字节的虚指针，再加对齐填充（x86-64 上指针 8 字节，4 字节的 `int` 后面补 4 字节） |
| 一个虚函数与四个虚函数一样大 | 对象里只放**一个**指针，函数地址在类共用的表里，加虚函数只加长表 |
| Cortex-M 上 4 字节变 8 字节 | 指针是 4 字节，`int` 4 字节，正好填满，没有填充 |
| `Empty` 与 `EmptyDerived` 都是 1 | 空类也要有地址，标准要求完整对象至少 1 字节；空的基类子对象可以被优化掉 |
| `EmptyMember` 是 8 | 空的**成员**优化不掉，它占 1 字节，后面为了对齐 `int` 补到 4，共 8 |
| `MI` 在 x86-64 上 24 字节 | 两个多态基类各带 8 字节虚指针，再加 `int m` 与填充 |
| `MI` 里 `Base2` 的偏移是 8（或 4） | 第二个基类子对象排在第一个之后，**不在对象开头** |

最后一条是第 3 节的全部内容所在。先把表打开。

> [!IMPORTANT]
> **对象里多出来的那一个指针叫虚指针（vptr），它指向的表叫虚函数表（vtable）。**
> 表是**每个类**一张，指针是**每个对象**一个。
> 表的内容与位置由 ABI 约定，标准没有规定；本章后面看到的每个偏移都来自具体实现。

## 1.2 空基类为什么能被优化掉

`EmptyDerived` 里那个 `Empty` 基类子对象没有占任何位置，
这不是「0 字节的对象」——标准不允许对象大小为 0——而是**空基类优化**：
基类子对象可以与派生类的其他成员共用地址。
它有一个直接后果：两个不同的子对象可能有**相同的地址**。

`C++`

```cpp
/* ebo.cpp    编译：g++ -std=c++17 ebo.cpp -o ebo */
#include <cstdio>

class Empty { };
struct Member  { Empty e; int v = 1; };           /* 空类当成员：占 1 字节 */
struct Derived : Empty { int v = 1; };            /* 空类当基类：可以被优化掉 */

int main() {
    Member m;
    Derived d;
    std::printf("sizeof(Member)  = %zu，e 的偏移 %td，v 的偏移 %td\n", sizeof(Member),
                reinterpret_cast<const char *>(&m.e) - reinterpret_cast<const char *>(&m),
                reinterpret_cast<const char *>(&m.v) - reinterpret_cast<const char *>(&m));
    std::printf("sizeof(Derived) = %zu，基类子对象地址 %p，对象地址 %p，%s\n", sizeof(Derived),
                static_cast<const void *>(static_cast<Empty *>(&d)),
                static_cast<const void *>(&d),
                (static_cast<const void *>(static_cast<Empty *>(&d)) == static_cast<const void *>(&d))
                    ? "同一个地址" : "不同地址");
    return 0;
}
```

`实测数据`
`Text`

```text
sizeof(Member)  = 8，e 的偏移 0，v 的偏移 4
sizeof(Derived) = 4，基类子对象地址 0000001c377ff894，对象地址 0000001c377ff894，同一个地址
```

`Derived` 只有 4 字节，基类子对象与派生类对象**地址完全相同**。
这一点在写「按地址比较两个子对象」的代码时会咬人：
两个 `dynamic_cast` 到不同基类的指针，地址可能一样，也可能差一个偏移，全看布局。

> [!WARNING]
> **不要用地址相等判断两个子对象是不是同一个对象。**
> 空基类优化会让不同子对象共用地址；虚继承会让同一个子对象在派生类里有固定偏移。
> 要比较身份，比较**指向最派生对象的指针**，或让类提供明确的比较接口。

## 1.3 标准管语义，ABI 管布局

标准对布局说了什么，值得看一眼原文，因为很多人的印象比实际严格：

`文档`

> "The object representation of an object of type T is the sequence of N unsigned char objects
> taken up by the object of type T, where N equals sizeof(T)."
>
> —— N4659 §6.9/4

标准把「对象表示」定义成 `sizeof(T)` 个 `unsigned char`，
但它并没有规定这些字节怎么排。真正规定排布的是 ABI。
在 x86-64 与 ARM 上，C++ 都用 **Itanium C++ ABI** 的布局规则；
MSVC 用的是另一套（`vptr` 的位置、`type_info` 的形态都不同）。
本章所有实测都出自 GCC，因此结论对 GCC 与 Clang 成立，
对 MSVC **待确认**。

`文档`

> "The result of sizeof applied to any other fundamental type (3.9.1) is implementation-defined."
>
> —— N4659 §8.3.3/1

一句话：**标准保证语义，不保证字节**。凡是本章给出的偏移量，
都要以「本机工具链的输出」的身份去用，不能写进跨编译器的协议里。

# 第 2 节 虚表里存什么、放在哪

## 2.1 把编译器的账本翻出来

GCC 有一个开关能把类布局转储出来：

`Bash`

```bash
g++ -std=c++17 -c -fdump-lang-class=layout.txt classes.cpp -o classes.o
```

MinGW 的 `g++` 不接受 `=文件名` 这种写法，转储会落到
`classes.cpp.001l.class` 里；内容与 Linux 侧一致。用它转储下面几个类：

`C++`

```cpp
/* classes.cpp   只用于看编译器的类布局转储，不运行：
   g++ -std=c++17 -c -fdump-lang-class=layout.txt classes.cpp -o classes.o */
class Base {
public:
    virtual void f();
    virtual void g();
    int a;
};

class Derived : public Base {
public:
    void f() override;
    virtual void h();
    int b;
};

class Base2 {
public:
    virtual void k();
};

class MI : public Base, public Base2 {
public:
    void f() override;
    void k() override;
    int m;
};

class VBase {
public:
    virtual void p();
    int v;
};

class VDerived : virtual public VBase {
public:
    void p() override;
    int d;
};

void Base::f() { }
void Base::g() { }
void Derived::f() { }
void Derived::h() { }
void Base2::k() { }
void MI::f() { }
void MI::k() { }
void VBase::p() { }
void VDerived::p() { }

int main() { return 0; }
```

## 2.2 表的四个槽位

转储里的行号是**相对表开头的字节偏移**，而对象里存的虚指针指向的是**表中间**：

`Text`

```
        _ZTV4Base 指向这里
              │
              ▼
        ┌───────────────┐  表内偏移 0
        │ 0             │  vptr[-2]：到对象开头的偏移（offset-to-top）
        ├───────────────┤  表内偏移 8
        │ &_ZTI4Base    │  vptr[-1]：type_info 对象的地址
        ├───────────────┤  表内偏移 16  ← 对象里的虚指针指向这里
        │ &Base::f      │  vptr[0]：第一个被声明的虚函数
        ├───────────────┤  表内偏移 24
        │ &Base::g      │  vptr[1]：第二个
        └───────────────┘
```

| 槽位 | 放什么 | 谁在用 |
|---|---|---|
| `vptr[-2]` | 到对象开头的偏移（`offset-to-top`） | `dynamic_cast` 从子对象找最派生对象 |
| `vptr[-1]` | 这个类的 `type_info` 对象地址 | `typeid`、`dynamic_cast`、异常匹配 |
| `vptr[0]`、`vptr[1]`…… | 虚函数地址，**按声明顺序**排 | 每一次虚调用 |
| `vptr[-3]` 及更前 | 虚基类偏移（只有带虚基类的类才有） | 访问虚基类的成员 |

第三条有例外：派生类覆盖基类虚函数时，**槽位顺序按基类的声明顺序**，
派生类新加的虚函数排在后面。`Derived` 的表就是这个样子：
16 号槽被换成 `Derived::f`，24 号槽仍是 `Base::g`，`Derived::h` 追加在 32 号。

`实测数据`
`Text`

```text
Derived::_ZTV7Derived: 5 entries
16    (int (*)(...))Derived::f      ← 覆盖，占原来 Base::f 的槽
24    (int (*)(...))Base::g         ← 没覆盖，还是基类那一个
32    (int (*)(...))Derived::h      ← 新加的，追加在后面
```

> [!IMPORTANT]
> **槽位顺序按基类的声明顺序定，不按派生类的写法定。**
> 这条规则的存在是为了让「一个编译好的基类函数体」在拿到任何派生类对象时都能用同一套偏移取到函数，
> 也就是 ABI 兼容性。它同时说明：**改动基类里虚函数的声明顺序，是二进制不兼容的改动**。

```cpp
                reinterpret_cast<const char *>(sub) - reinterpret_cast<const char *>(obj),

                reinterpret_cast<const char *>(vp[0]) - reinterpret_cast<const char *>(vp));

                sizeof(A), sizeof(B), sizeof(C), sizeof(D));

    row("Derived", &d, &d, &typeid(Derived));
    row("MI/Base 部分", static_cast<Base *>(&mi), &mi, &typeid(MI));
    row("MI/Base2 部分", static_cast<Base2 *>(&mi), &mi, &typeid(MI));
    row("D 主 vptr", &dd, &dd, &typeid(D));
```

## 3.3 `static_cast` 与 `dynamic_cast` 差在哪

| 写法 | 编译期做什么 | 运行期做什么 | 什么时候能用 |
|---|---|---|---|
| `static_cast<Base2 *>(p)` | 已知 `p` 的实际类型是 `Both`，直接把偏移加上 | 什么也不做 | 编译期就能确定「转得过去」 |
| `dynamic_cast<Base2 *>(p)` | 生成一次对 `__dynamic_cast` 的调用 | 读 `vptr[-1]` 拿类型信息，沿基类表查找，必要时调整指针 | 只知道 `p` 是某个多态基类，实际类型未知 |

把基类指针转成派生类指针时，两者的差别更明显：

`C++`

```cpp
/* downcast.cpp    编译：g++ -std=c++17 downcast.cpp -o downcast */
#include <cstdio>

class Base1 { public: virtual ~Base1() = default; int a = 1; };
class Base2 { public: virtual void g() { } int b = 2; };
class Both  : public Base1, public Base2 { public: int c = 3; };

int main() {
    Both obj;
    Base1 *p1 = &obj;

    /* static_cast 反着把偏移减掉，不做任何检查 */
    Both *s = static_cast<Both *>(p1);
    std::printf("static_cast<Both *> 得到 %p（原对象 %p）\n",
                static_cast<void *>(s), static_cast<void *>(&obj));

    /* 下面一行把 Base1 换成了 Base2：这是交叉转换，static_cast 转不过去，
       编译器直接拦下，错误原文见正文 */
    return 0;
}
```

`Base2` 与 `Base1` 之间没有继承关系，`static_cast` 转不过去，编译器会说：

`实测数据`
`Text`

```text
error: 'static_cast' from 'Base1*' to 'Base2*' requires a dynamic_cast
```

而 `dynamic_cast` 能完成这次交叉转换，代价是第 5 节要量的一次类型信息查找。

## 3.4 写错了会怎样：用 `reinterpret_cast` 穿越多继承

`static_cast` 会把偏移算进去。换成 `reinterpret_cast`，**偏移不加上**，
指针指向对象开头，但类型却说是 `Base2 *`：

`C++`

```cpp
/* badcast.cpp    编译：g++ -std=c++17 -O0 badcast.cpp -o badcast */
#include <cstdio>

class Base1 {
public:
    virtual void f() { }
    int a = 111;
};

class Base2 {
public:
    virtual void g() { }
    int b = 222;
};

class Both : public Base1, public Base2 {
public:
    int c = 333;
};

/* Base2 子对象不在对象开头，两种写法给出的地址完全不同 */
static void look(const char *how, Base2 *p) {
    std::printf("%-22s 指针 %p  b = %d\n", how, static_cast<void *>(p), p->b);
}

int main() {
    Both obj;
    std::printf("对象开头 %p，Base1 部分 %p，Base2 部分 %p\n",
                static_cast<void *>(&obj),
                static_cast<void *>(static_cast<Base1 *>(&obj)),
                static_cast<void *>(static_cast<Base2 *>(&obj)));

    look("static_cast（对）", static_cast<Base2 *>(&obj));
    look("reinterpret_cast（错）", reinterpret_cast<Base2 *>(&obj));

    std::printf("\n再看经 void* 中转：\n");
    void *raw = static_cast<Base1 *>(&obj);
    look("static_cast<void*> 再 static_cast", static_cast<Base2 *>(raw));
    return 0;
}
```

`实测数据`
`Text`

```text
对象开头 0000005cb47ff990，Base1 部分 0000005cb47ff990，Base2 部分 0000005cb47ff9a0
static_cast（对）   指针 0000005cb47ff9a0  b = 222
reinterpret_cast（错） 指针 0000005cb47ff990  b = 111

再看经 void* 中转：
static_cast<void*> 再 static_cast 指针 0000005cb47ff990  b = 111
```

**程序没有崩，它安静地读错了地方。** `reinterpret_cast` 拿到的指针比正确的少了 16 字节，
`p->b` 按 `Base2` 的布局去偏移 8 字节读，读到的是 `Base1` 的 `a`。
如果 `Base1` 与 `Base2` 的成员类型不同，读出来的还会是一段无意义的位模式。

经 `void *` 中转的那一行同样是错的：`static_cast<Base1 *>` 得到的是 `Base1` 子对象的地址，
把它当 `void *` 存起来再转成 `Base2 *` 时，编译器**没有任何信息**可以判断需要加偏移，
只能原样转换。**C++ 里唯一正确的做法是从最派生类型出发做一次 `static_cast`，
或者用 `dynamic_cast`。**

> [!CAUTION]
> **`reinterpret_cast` 只在两个类型之间做「按位重新解释」，它不碰指针的值。**
> 只要涉及多继承、虚继承或任何子对象不在偏移 0 的情况，用它做向上转换就是错的，
> 而且多数时候不会崩，只是在错误的位置读数据。
> 需要跨类型转换时用 `static_cast`；编译期不知道实际类型时用 `dynamic_cast`。

# 第 4 节 虚继承的布局

## 4.1 两份基类子对象的问题

《05-类与面向对象/07-继承.md》第 6.3 小节讲过菱形继承：
`D` 继承 `B` 与 `C`，而 `B`、`C` 又都继承 `A`，那么 `D` 里会出现**两份** `A` 子对象，
访问 `d.a` 就成了歧义。第 6.4 小节给出的解法是虚继承：

`C++`

```cpp
/* diamond.cpp    编译：g++ -std=c++17 diamond.cpp -o diamond */
#include <cstdio>

class A { public: int a = 1; };
class B : virtual public A { public: int b = 2; };
class C : virtual public A { public: int c = 3; };
class D : public B, public C { public: int d = 4; };

int main() {
    D obj;
    obj.a = 9;                       /* 虚继承之后这一句不再歧义 */
    std::printf("sizeof(D) = %zu，obj.a = %d\n", sizeof(obj), obj.a);
    std::printf("A 子对象只有一个：%p\n", static_cast<void *>(static_cast<A *>(&obj)));
    std::printf("B 子对象 %p，C 子对象 %p\n",
                static_cast<void *>(static_cast<B *>(&obj)),
                static_cast<void *>(static_cast<C *>(&obj)));
    return 0;
}
```

`实测数据`
`Text`

```text
sizeof(D) = 48，obj.a = 9
A 子对象只有一个：0000006b1f5ffa30
B 子对象 0000006b1f5ffa30，C 子对象 0000006b1f5ffa40
```

`a` 只有一个，但**代价是布局复杂了**：48 字节，`A` 子对象排在最后。
原因很直接：虚基类的位置必须在**所有**派生类里一致，
而 `B` 与 `C` 单独使用时也要能放下同一个 `A`，
编译器只能把 `A` 推到最派生对象布局的末尾，再用一张偏移表在运行期找到它。

`B` 单独看是这样的：

`实测数据`
`Text`

```text
sizeof(B) = 32（只含一个 int b 与一个虚基类 A）
B 里 A 的偏移 = 16
B 主 vptr 的 vptr[-3] = 16
```

（这几行由 `vbase1.cpp` 打出，源码与 `vbases.cpp` 结构相同，见附录 A。）

## 4.2 虚基类偏移槽

`vptr[-3]` 是虚基类偏移槽。**它的值是「从这个虚指针的地址点到虚基类子对象的偏移」。**
四个数据点可以验证这条规则：

`C++`

```cpp
/* vbases.cpp   编译：g++ -std=c++17 vbases.cpp -o vbases
   看每个子对象的 vptr[-3]（虚基类偏移槽）与真实偏移之间的关系 */
#include <cstdio>
#include <cstdint>
#include <cstddef>

class A { public: virtual void a() { } int x = 1; };
class B : virtual public A { public: virtual void b() { } int y = 2; };
class C : virtual public A { public: virtual void c() { } int z = 3; };
class D : public B, public C { public: virtual void d() { } int w = 4; };

using Slot = void (*)();
using Vptr = Slot *;

static void show(const char *name, const void *sub, const void *obj) {
    const Vptr vp = *reinterpret_cast<const Vptr *>(sub);
    std::printf("%-14s 子对象偏移 %-4td  vptr[-3]（虚基偏移槽）= %-5td  vptr[-2]（到顶）= %td\n",
                name,
                reinterpret_cast<const char *>(sub) - reinterpret_cast<const char *>(obj),
                static_cast<std::ptrdiff_t>(reinterpret_cast<const std::intptr_t *>(vp)[-3]),
                static_cast<std::ptrdiff_t>(reinterpret_cast<const std::intptr_t *>(vp)[-2]));
}

int main() {
    D d;
    std::printf("sizeof(D) = %zu\n", sizeof(D));
    std::printf("D 里 A 的真实偏移 = %td\n\n",
                reinterpret_cast<const char *>(static_cast<A *>(&d)) -
                reinterpret_cast<const char *>(&d));
    show("D 主 vptr", &d, &d);
    show("D/B 子对象", static_cast<B *>(&d), &d);
    show("D/C 子对象", static_cast<C *>(&d), &d);
    show("D/A 子对象", static_cast<A *>(&d), &d);
    return 0;
}
```

`实测数据`
`Text`

```text
sizeof(D) = 48
D 里 A 的真实偏移 = 32

D 主 vptr     子对象偏移 0     vptr[-3]（虚基偏移槽）= 32     vptr[-2]（到顶）= 0
D/B 子对象  子对象偏移 0     vptr[-3]（虚基偏移槽）= 32     vptr[-2]（到顶）= 0
D/C 子对象  子对象偏移 16    vptr[-3]（虚基偏移槽）= 16     vptr[-2]（到顶）= -16
D/A 子对象  子对象偏移 32    vptr[-3]（虚基偏移槽）= 0      vptr[-2]（到顶）= -32
```

`A` 的真实偏移是 32，四行的规律非常整齐：

| 子对象 | 该虚指针的地址点 | 到 `A` 的距离 | `vptr[-3]` |
|---|---|---|---|
| 主虚指针（对象开头） | 0 | 32 | 32 |
| `B` 子对象与主虚指针同一地址 | 0 | 32 | 32 |
| `C` 子对象 | 16 | 32 - 16 = 16 | 16 |
| `A` 子对象 | 32 | 32 - 32 = 0 | 0 |

`vptr[-2]` 一列也对得上：到对象开头的偏移分别是 0、0、-16、-32。

> [!IMPORTANT]
> **带虚基类的类，虚表在 `vptr[-1]` 之前多出一段「虚基类偏移」槽，一行一个虚基类。**
> 编译器要访问 `A` 的成员时，先读这个槽，把偏移加到当前子对象地址上，再访问。
> **多一层间接，而且这个偏移在编译期是不知道的**——它取决于对象最终是哪个派生类。

`vtour.cpp` 里 `D` 那三行同样是这套规则：
`D 主 vptr` 的 `vptr[-3]` 是 32，与「`D` 里 `A` 的真实偏移 = 32」正好相等。
**这不是巧合，是同一个数。** 编译器生成的取虚基类成员的代码就是这么算的。

## 4.3 访问虚基类成员要多走几步

因为偏移要运行期才知道，代码里多出一次取表：

`实测数据`
`Assembly`

```asm
_Z11read_vbaseP1B:
        ldr     r3, [r0]          @ 取虚指针
        ldr     r3, [r3, #-12]    @ 取 vptr[-3]：虚基类偏移
        ldr     r0, [r0, r3]      @ 加上偏移，才读到 A 的成员
        bx      lr
```

（节选，Cortex-M3 `-O2` 生成的形状。）

对比不含虚基类的版本，直接一次 `ldr` 就读到了：

| 情形 | 取一个 `int` 成员的指令条数 |
|---|---|
| 普通继承 | 1 条 `ldr`（偏移编译期已知） |
| 虚继承 | 2 条 `ldr` 取偏移，1 条 `ldr` 读成员，共 3 条 |

这条链还带来一个后果：**虚基类的偏移表是常量，但 `vptr[-3]` 的读取不能被提到循环外面**，
因为 `vptr` 本身可能因为对象实际类型不同而不同。

## 4.4 虚继承在表里还留下了什么

`-fdump-lang-class` 对 `VDerived` 的输出里，除了虚基类偏移槽，还有两样东西：

`实测数据`
`Text`

```text
Vtable for VDerived
VDerived::_ZTV8VDerived: 8 entries
0     16
8     (int (*)(...))0
16    (int (*)(...))(& _ZTI8VDerived)
24    (int (*)(...))VDerived::p
32    18446744073709551600
40    (int (*)(...))-16
48    (int (*)(...))(& _ZTI8VDerived)
56    (int (*)(...))VDerived::_ZTv0_n24_N8VDerived1pEv

VTT for VDerived
VDerived::_ZTT8VDerived: 2 entries

```

| 名称 | 是什么 | 谁在用 |
|---|---|---|
| `_ZTV8VDerived` | 虚表，注意它有**两段**，第二段属于虚基类 `VBase` 子对象 | 每一次虚调用 |
| `_ZTv0_n24_N8VDerived1pEv` | **虚 thunk**：`_ZTv0_` 表示「从虚表偏移 0 处取虚基类偏移，调整 this」 | 通过虚基类指针调用 `p()` |
| `_ZTT8VDerived` | **VTT**（virtual table table）：两段虚表的地址表 | 构造函数与析构函数，它们要在构造过程中切换虚指针 |

构造函数里会看到对 VTT 的引用，因为「虚指针指向哪一段」在构造过程中要变化：
先指向基类那一段，构造到本类时再换成自己那一段。
这是《06-更底层/08-CRT 与程序启动.md》第 2 节里全局对象构造要用到的机制。

> [!NOTE]
> 这一节把 `05-类与面向对象/07-继承.md` 第 6 节留下的账结清了。
> 语义上「共同的基类只留一份」，实现上是：**虚基类子对象固定排在可选部分，
> 位置由虚表里的偏移槽记录，访问时多一层取表**。
> 代价是对象更大（`D` 48 字节，非虚继承的版本 40 字节）、代码更长、布局跨编译器差异更大。
> 反过来说，**能不用虚继承就不用**：它在语言里存在，是为了解决菱形继承这一个问题。

## 5.2 三类 `type_info`

`type_info` 不是单独一个类，而是一个小家族。**第一槽（`type_info` 对象自己的虚指针）
指向哪一张表，决定了 `dynamic_cast` 走哪条算法**：

`实测数据`
`Text`

```text
_ZTI4Base     → __class_type_info         （没有基类）
_ZTI7Derived  → __si_class_type_info      （单继承，si = single inheritance）
_ZTI2MI       → __vmi_class_type_info     （多继承，vmi = virtual/multiple inheritance）
_ZTI1D        → __vmi_class_type_info     （菱形）
```

后两类的对象比前两类多出两个字段和一张基类表。`MI` 的那份：

`实测数据`
`Text`


| 偏移 | 值 | 字段 | 读法 |
|---|---|---|---|
| 0 | `__vmi_class_type_info` 的虚表 | 继承自 `type_info` | 决定走哪条算法 |
| 8 | `_ZTS2MI` | 类型名 | `"2MI"` |
| 16（低 4 字节） | 0 | `__flags` | 既不是菱形也不是重复基类 |
| 16（高 4 字节） | 2 | `__base_count` | 有两个基类 |

`__flags = 2` 表示「菱形」，`__base_count = 2`，

## 5.3 `dynamic_cast` 一次要做多少事

把三类放在一起看，`dynamic_cast<Derived *>(p)` 的动作是：

`Text`

```
   1. 从 p 指的位置读 vptr
   2. 读 vptr[-1] 拿到 type_info
   3. 看 type_info 的第一槽，选算法
        __class_type_info        → 只有一种可能：比名字
        __si_class_type_info     → 沿基类链往上走
        __vmi_class_type_info    → 遍历 __base_info[]，每个基类递归
   4. 找到以后，按记录下来的偏移把指针调整到派生类子对象
```

第 3 步里的「递归遍历基类表」就是它贵的根源：
**每一次 `dynamic_cast` 都在走一棵运行期才知道形状的树**，
而这棵树在编译期是完全静态的、编译器帮不上忙。

## 5.4 代价实测

四组操作各跑两千万次，指针表里混着四种类型，编译器无法把查找提到循环外：

`C++`

```cpp
/* dcost.cpp    编译：g++ -std=c++17 -O2 dcost.cpp -o dcost */
#include <cstdio>
#include <chrono>

class Base {
public:
    virtual ~Base() = default;
    virtual int v() const { return 1; }
};

class Derived : public Base {
public:
    int v() const override { return 2; }
    int extra = 42;
};

class Other : public Base {
public:
    int v() const override { return 3; }
};

class SiblingA : public Base { };
class SiblingB : public Base { };

constexpr long kN = 20000000;      /* 两千万次 */
constexpr int kMix = 64;           /* 指针表里混着四种类型，编译器无法把查找提到循环外 */

static Base *g_mix[kMix];
volatile long g_sink;              /* 防止整个循环被优化掉 */

template <class F>
static double time_ns(F body) {
    auto t0 = std::chrono::steady_clock::now();
    body();
    auto t1 = std::chrono::steady_clock::now();
    return std::chrono::duration<double, std::nano>(t1 - t0).count() / kN;
}

int main() {
    static Derived  ders[kMix / 4];
    static Other    others[kMix / 4];
    static SiblingA sibas[kMix / 4];
    static SiblingB sibbs[kMix / 4];
    for (int i = 0; i < kMix; i += 4) {
        g_mix[i + 0] = &ders[i / 4];
        g_mix[i + 1] = &others[i / 4];
        g_mix[i + 2] = &sibas[i / 4];
        g_mix[i + 3] = &sibbs[i / 4];
    }

    long acc = 0;

    double t_virtual = time_ns([&] {
        for (long i = 0; i < kN; ++i) { acc += g_mix[i & (kMix - 1)]->v(); }
    });

    double t_static = time_ns([&] {
        for (long i = 0; i < kN; ++i) {
            Derived *q = static_cast<Derived *>(g_mix[i & (kMix - 1)]);  /* 只加偏移 */
            acc += q->extra;
        }
    });

    double t_dyn_ok = time_ns([&] {
        for (long i = 0; i < kN; ++i) {
            Derived *q = dynamic_cast<Derived *>(g_mix[i & (kMix - 1)]);  /* 四次里中一次 */
            acc += (q != nullptr) ? q->extra : 0;
        }
    });

    double t_dyn_cross = time_ns([&] {
        for (long i = 0; i < kN; ++i) {
            SiblingB *q = dynamic_cast<SiblingB *>(g_mix[i & (kMix - 1)]);  /* 交叉转换 */
            acc += (q != nullptr) ? 1 : 0;
        }
    });

    g_sink = acc;
    std::printf("循环次数 %ld，指针表里四种类型轮流\n", kN);
    std::printf("虚函数调用                         %8.3f ns/次\n", t_virtual);
    std::printf("static_cast + 读成员               %8.3f ns/次\n", t_static);
    std::printf("dynamic_cast 到 Derived（四中一）   %8.3f ns/次\n", t_dyn_ok);
    std::printf("dynamic_cast 交叉到 SiblingB（四中一）%8.3f ns/次\n", t_dyn_cross);
    std::printf("累计结果 %ld\n", static_cast<long>(g_sink));
    return 0;
}
```

`实测数据`

三次运行的重复性：

`实测数据`

| 运行 | 虚函数调用 | `static_cast` | `dynamic_cast` 到 `Derived` | 交叉转换 |
|---|---|---|---|---|
| 第一次 | 0.921 ns | 0.268 ns | 11.294 ns | 11.591 ns |
| 第二次 | 0.968 ns | 0.266 ns | 11.313 ns | 11.647 ns |
| 第三次 | 0.901 ns | 0.278 ns | 11.698 ns | 11.495 ns |

**一次 `dynamic_cast` 相当于十次虚调用。** 它不只是「多一次间接跳转」，
而是十几次内存访问加一棵运行期才展开的继承树遍历。
交叉转换与向下转换的价格差不多，因为两者都要把类型信息走一遍。

> [!TIP]
> **`dynamic_cast` 是补丁，不是设计。**
> 《05-类与面向对象/08-多态：重载、虚函数与它们的分工.md》第 8.1 小节
> 从设计角度说过这句话；这里补上另一半依据：
> 它在热循环里每次要花十几纳秒，是虚调用的十倍。
> 调用点上一旦出现一串 `dynamic_cast` 加 `if`，性能与可维护性会同时变差。

# 第 6 节 在 MCU 上这些代价意味着什么

## 6.1 同一份代码，五种写法

单片机上的约束与桌面不同：flash 常常只有 64 KB 到 512 KB，
RAM 只有 20 KB 到 128 KB，多出来的每一 KB 都要从别处省。
下面这份程序用宏 `SHAPE` 选出五种写法算同一件事（按类型求面积），
其余部分完全相同，因此 `size` 的差就是写法的差：

`C++`

```cpp
/* Cortex-M 侧（在 WSL 里交叉编译 + QEMU 真跑）：
   同一份源码，用 -DSHAPE= 选出五种写法，各编一个 .elf 比体积：

   arm-none-eabi-g++ -mcpu=cortex-m3 -mthumb -O2 -nostartfiles --specs=nosys.specs \
     -DSHAPE=0 -T STM32F103XX_FLASH.ld startup_stm32f103xe.s mcu_shape.cpp -o shape0.elf
   ……把 -DSHAPE=0 依次换成 1、2、3、4 得到 shape1..shape4.elf；
   再加两组关掉运行时支持的：
   …… -DSHAPE=3 -fno-rtti … -o shape3_nortti.elf
   …… -DSHAPE=4 -fno-exceptions … -o shape4_noexc.elf
   然后：
   arm-none-eabi-size shape*.elf
   arm-none-eabi-objdump -h shape2.elf */
#include "semihosting.h"

#if SHAPE == 0
/* 写法 0：一个 kind 字段加 switch，最像 C 的写法 */
struct Shape { int kind; int a; };
static int area(const Shape &s) {
    switch (s.kind) {
    case 0:  return 3 * s.a * s.a;
    default: return s.a * s.a;
    }
}
static int run() {
    const Shape arr[2] = { {0, 3}, {1, 4} };
    int sum = 0;
    for (int i = 0; i < 2; ++i) { sum += area(arr[i]); }
    return sum;
}

#elif SHAPE == 1
/* 写法 1：结构体里放一个函数指针 */
struct Shape { int (*area)(const Shape *); int a; };
static int circle_area(const Shape *s) { return 3 * s->a * s->a; }
static int square_area(const Shape *s) { return s->a * s->a; }
static int run() {
    const Shape arr[2] = { {circle_area, 3}, {square_area, 4} };
    int sum = 0;
    for (int i = 0; i < 2; ++i) { sum += arr[i].area(&arr[i]); }
    return sum;
}

#else
/* 写法 2 起：虚函数。对象里多一个虚指针，表放在只读段 */
class Shape {
public:
    Shape(int a) : a_(a) { }
    virtual ~Shape() { }
    virtual int area() const { return a_ * a_; }
    int a_;
};

class Circle : public Shape {
public:
    explicit Circle(int r) : Shape(r) { }
    int area() const override { return 3 * a_ * a_; }
};

class Square : public Shape {
public:
    explicit Square(int s) : Shape(s) { }
    int area() const override { return a_ * a_; }
};

/* 对象与指针表都放在全局；指针表加 volatile，
   免得编译器把「刚写进去的地址」传播过来，把虚调用改成直接调用 */
static Circle  g_circle(3);
static Square  g_square(4);
static Shape  *volatile g_shapes[2];

static int run() {
    g_shapes[0] = &g_circle;
    g_shapes[1] = &g_square;
    int sum = 0;
    for (int i = 0; i < 2; ++i) { sum += g_shapes[i]->area(); }
    return sum;
}
#endif

extern "C" void SystemInit(void) { }
extern "C" void _init(void) { }

extern "C" {
/* -nostartfiles 去掉了 crtbegin.o / crtend.o，它们本来负责定义 __dso_handle 与 _fini；
   只要文件里有带构造函数的全局对象，析构注册就会引用到这两个名字，因此自己补上。
   删掉这一段会得到：undefined reference to `__dso_handle' 与 undefined reference to `_fini' */
void *__dso_handle = nullptr;
void _fini(void) { }
}

int main(void) {
    sh_row("area 合计 = ", static_cast<uint32_t>(run()));
    for (;;) { }
}
```

（`-DSHAPE=3` 与 `4` 在本文件里被裁掉了，完整版本见附录 A。）

编译时加上 `-ffunction-sections -fdata-sections -Wl,--gc-sections`，
让链接器把没人引用的函数与数据丢掉——这是真实固件工程的常规做法：

`Bash`

```bash

  -ffunction-sections -fdata-sections -Wl,--gc-sections \
  -T STM32F103XX_FLASH.ld -DSHAPE=2 startup_stm32f103xe.s mcu_shape.cpp -o gc_shape2.elf
arm-none-eabi-size gc_shape0.elf gc_shape1.elf gc_shape2.elf gc_shape3.elf gc_shape4.elf
```

`实测数据`
`Text`

```text

    732	      0	   1536	   2268	    8dc	gc_shape0.elf
    836	      0	   1536	   2372	    944	gc_shape1.elf
```

| 写法 | flash（`text`） | RAM 初值（`data`） | RAM 清零（`bss`） |
|---|---|---|---|

| 1：函数指针 | 836 | 0 | 1536 |

从 836 到 9212，**涨了 8376 字节，是函数指针写法的十倍**；
`data` 段从 0 变成 1336，也就是**上电时要从 flash 搬 1336 字节进 RAM**。

> [!IMPORTANT]
> **在 MCU 上，虚函数的开销主要不在「每个对象多一个指针」，而在被拉进来的整个 C++ 运行时。**
> 对象里的那个指针只值 4 字节；被带进来的 malloc、stdio、栈展开表、`atexit` 注册
> 加起来是几 KB 的 flash 与一千多字节的 RAM。

## 6.3 两个开关的效果

| 开关 | flash 变化 | 说明 |
|---|---|---|
| `-fno-rtti` | 9212 → 7880，**省 1332 字节** | 不再生成 `type_info` 与查表代码；代价是 `typeid` 与 `dynamic_cast` 都不能用了 |
| `-fno-exceptions` | 9212 → 9212，**一个字节也没省** | 展开表在 ARM EABI 上是默认生成的（用于回栈），关掉异常并不去掉它 |

第二行值得记住：**「关掉异常能省很多空间」这句话在 ARM 裸机上不成立**，
至少在这个工具链与这套选项下不成立。
`shape4`（真的用了 `try`/`catch`）是 15664 字节，比 `shape2` 多 6452 字节，
这部分才是异常运行时的真实代价。

> [!WARNING]
> **在 ARM 裸机上不要靠 `-fno-exceptions` 省体积。**
> 要省就从 `-fno-rtti` 与 `-fno-use-cxa-atexit` 入手，
> 并且用 `size` 前后各量一次——体积这件事只能靠测，不能靠猜。

## 6.4 写错了会怎样：`__dso_handle` 与 `_fini`

上面那份程序里有一段补 `__dso_handle` 与 `_fini` 的代码。
把它删掉，只要文件里有带构造函数的全局对象，链接就会失败：

`实测数据`
`Text`

```text
mcu_shape.cpp:(.text.startup+0xb8): undefined reference to `__dso_handle'
t2.elf: hidden symbol `__dso_handle' isn't defined
/usr/lib/.../libc.a(libc_a-fini.o): in function `__libc_fini_array':
undefined reference to `_fini'
collect2: error: ld returned 1 exit status
```

原因是 `-nostartfiles` 把 `crtbegin.o` 与 `crtend.o` 也去掉了，
而这两个文件恰好负责定义 `__dso_handle` 与 `_fini`。
桌面程序里它们由启动文件提供，裸机上要么补定义，要么改用完整启动文件。

**没有全局对象时不会报这个错**：`shape0` 与 `shape1` 都没有构造函数，
删掉那段照样链接通过。这也解释了为什么「加了一个全局对象，工程突然链接不过」
这类问题在嵌入式项目里会出现——问题不在那一行对象定义，
而在它悄悄启用了析构注册这条路径。

## 6.5 在 MCU 上什么时候用虚函数

| 情形 | 建议 |
|---|---|
| flash 很紧（几十 KB）、又用不到多态 | 用 `switch` 或函数指针表，成本是两个数量级之差 |
| 确实需要运行期多态，且对象数量不多 | 虚函数可以用，但要`-fno-rtti`，并确认没有全局对象 |
| 对象数量很大（成百上千） | 先算 RAM：每个对象多一个指针，在 20 KB RAM 的芯片上很快会见底 |
| 用 `dynamic_cast` 做类型分发 | 换成虚函数。在 MCU 上一次十几纳秒、几 KB 代码，代价过高 |
| 需要与 C 代码互操作、按字节序列化 | 不要用虚函数：对象开头那几个字节会破坏两边约定好的布局 |

反过来说，**「虚函数在单片机上太贵」这句话只对了一半**：
贵的是被拉进来的运行时，不是那一个指针。
把 `-fno-rtti` 加上、把全局对象去掉、把虚析构换成非虚析构，
许多工程里虚函数的净开销会落到 1 KB 以内。

## 6.6 硬件浮点的代价与边界

第 6.1 小节测的是 C++ 机制在 flash 里占多少，这一小节换到算术上问同一件事。
Cortex-M4F 里的 FPU 是一个可选单元：同一份源码把它打开或关上，编译出来的指令完全不同，
而语言层面看不出差别。

对照只有一个变量：`-mfloat-abi` 取 `hard` 还是 `soft`。源码、优化等级、链接脚本、
时钟与计时办法全部相同，因此两列的差就是这个选项的差。

`实测数据`

| 条件 | 取值 |
|---|---|
| 芯片 | STM32G431（Cortex-M4F r0p1，单精度走硬件 FPU） |
| 时钟 | HSI 16 MHz，复位后的默认值，不配 PLL |
| 优化等级 | `-O2` |
| 每段循环的迭代次数 | 20,000 次 |
| 计时办法 | 读 `DWT->CYCCNT` 两次相减，再除以迭代次数 |
| 两列的唯一差别 | `-mfloat-abi=hard` 与 `-mfloat-abi=soft`，两列的 `-mfpu` 都是 `fpv4-sp-d16` |
| 工具链 | 同一套 `arm-none-eabi-gcc` 13.2.1（WSL 的 Ubuntu 里），同一个 Makefile |

`实测数据`

| 每次迭代的周期数 | `hard`（走 FPU） | `soft`（软件例程） | 倍数 |
|---|---:|---:|---:|
| `int` 加法 | 9.00 | 9.00 | 1 倍 |
| `int` 除常量 3（编译期换成魔数乘法） | 14.92 | 14.92 | 1 倍 |
| **`float` 乘加** | **11.00** | **100.00** | **约 9 倍** |
| `double` 乘加 | 166.00 | 166.00 | 1 倍 |

**四行里只有一行动了。** `int` 的两行与 `double` 那一行逐位相同，
这正是变量隔离在做的事：变的只有单精度那条路径，不是整个程序的代码生成。
若四行一起变，就说明还有别的东西跟着动了，那一组数字不能用来归因于 FPU。

`double` 那一行还会跳 1 个周期：同一段代码重编一次，它会在 165 与 166 之间变化，
原因是代码在 flash 里的位置变了，取指路径随之改变。1 个周期量级的变化不适合当作结论。

**`int` 除常量 3 这一行名不副实**：除数是编译期常量时，gcc 把「除以 3」换成了魔数乘法加符号修正，
两块板都一样，因此这一行测到的是 64 位乘法与循环本身的快慢，不是除法的快慢。

```c
           --specs=rdimon.specs main.c startup_stm32g431xx.s -o g431_hard.elf
   把 -mfloat-abi=hard 换成 soft 得到另一份产物，其余选项一字不改。 */
#define SCB_CPACR (*(volatile uint32_t *)0xE000ED88u)   /* 协处理器访问控制 */

void SystemInit(void)
{
    SCB_CPACR |= (0xFu << 20);      /* CP10 与 CP11 允许全访问 */
}
```

`实测数据`
`Text`

```text
CPUID    = 0x410fc241
CPACR    = 0x00f00000  (CP10/CP11 全访问 = 0x00F00000)
```

`CPUID` 的 partno 字段 `0xC24` 就是 Cortex-M4；`CPACR` 读回 `0x00f00000`，说明那次写生效了。

**两个选项分别管什么**：

| 选项 | 管什么 |
|---|---|
| `-mfpu=fpv4-sp-d16` | 允许生成哪些浮点指令：`fpv4` 是这一代内核的浮点版本，`sp` 表示只有单精度，`d16` 表示 16 个双字寄存器（`s0` 到 `s31`） |
| `-mfloat-abi=hard` | 生成的代码用 FPU 指令，参数与返回值也走 FPU 寄存器 |
| `-mfloat-abi=soft` | 生成的代码调用软件例程，参数与返回值走整数寄存器 |

`d16` 这个后缀就是「没有双精度」在编译选项上的写法：`fpv4-sp-d16` 里没有双精度的位置，
`double` 因此只能落到软件例程上。

**混用风险**：`-mfloat-abi` 改的是 ABI，不是某个文件的局部开关，
因此一份固件里的所有目标文件与静态库必须同为 `hard`。这条规则由链接器把关，
在主机上就能验证：

`C`

```c
/* lib.c   库这一侧：一个函数，编译成 lib_soft.o 与 lib_hard.o */
float scale(float x) { return x * 2.0f; }
```

`C`

```c
/* app.c   主程序这一侧：调用库函数，编译成 app_hard.o */
extern float scale(float);
volatile float y;
int main(void) { y = scale(1.5f); for (;;) { } }
```

`实测数据`
`Bash`

```bash
# 库用 soft 编，主程序用 hard 编，然后链接
arm-none-eabi-gcc -c -mcpu=cortex-m4 -mthumb -O2 -mfpu=fpv4-sp-d16 -mfloat-abi=soft lib.c -o lib_soft.o
arm-none-eabi-gcc -c -mcpu=cortex-m4 -mthumb -O2 -mfpu=fpv4-sp-d16 -mfloat-abi=hard app.c -o app_hard.o
arm-none-eabi-gcc -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard \
  -nostartfiles -Wl,--gc-sections app_hard.o lib_soft.o -o app.elf
```

`实测数据`
`Text`

```text
ld: error: app.elf uses VFP register arguments, lib_soft.o does not
ld: failed to merge target specific data of file lib_soft.o
collect2: error: ld returned 1 exit status
```

把 `lib_soft.o` 换成用 `-mfloat-abi=hard` 编出来的同一份库，链接直接通过。
**预编译库的 ABI 是定死的**，主程序改成 `hard` 而库还是 `soft` 时，
工程会停在链接这一步，报出来的就是上面那两行。

> [!IMPORTANT]
> **选了 FPU 不等于白拿。**
> 它只对单精度便宜：`float` 乘加从 100 周期降到 11 周期，
> 而 `int` 两行与 `double` 那一行一个周期都没省。
> `double` 在这颗内核上没有硬件，仍然走与 `soft` 版相同的软件例程，
> 因此「有 FPU 就可以放心用 `double`」不成立；
> 把中间量留在 `float` 上，那 9 倍才吃得到。

换一块没有 FPU 的板子跑同一份源码，四行数字就成了两块内核的对照。

`实测数据`

| 周期/次（`-O2`，n=20000） | F103（Cortex-M3 r1p1，HSI 8 MHz） | G431（Cortex-M4F，HSI 16 MHz） |
|---|---:|---:|
| `int` 加法 | 8.00 | 9.00 |
| `int` 除常量 3（魔数乘法） | 17.15 | 14.92 |
| `float` 乘加 | **100.00** | **11.00** |
| `double` 乘加 | 178.00 | 166.00 |
| `int` 除运行期除数（`volatile`） | **19.61** | **待补** |

**两块板的时钟不同，这是读这张表最要紧的一句话。** 两块板只换 `-mcpu` 与浮点选项，其余相同；
周期数只反映指令与内核结构，换频率不变，两列可以直接并排比，墙钟不可比；F103 的频率用哨兵加
`mdw` 测出：baseline 5.090 s、工作版 13.123 s 与 64,000,006 周期，对应 **7.97 MHz**，与 HSI 的 8 MHz 相符（偏 −0.4%）。

**唯一能直接证明「有 FPU 快多少」的是 `float` 那一行：100.00 对 11.00。** `double` 的 178 与 166 不是
FPU 造成的：两边都走软件例程，差的 12 个周期来自 M4 的 DSP 指令（`SMLAL` 一类）与更快的 64 位乘法；
旁证是 G431 用 `-mfloat-abi=soft` 编出来的同一段循环也是 100.00。`int` 加法那一行 M3 反而快 1 个周期。

**「Cortex-M3 没有硬件除法」不成立。** 这块 F103C8 的核是 Cortex-M3 r1p1（`CPUID = 0x411fc231`，
partno `0xC23`），实测**有** `sdiv` 与 `udiv`，三个运行期除法的结果全对：
`1000000 / 7 = 142857`、`-1000000 / 7 = -142857`、`4000000000 / 7 = 571428571`，对应下面三条指令。

`实测数据`
`Text`

```text
0x0800709c:  sdiv  r1, r1, r3
0x080070ae:  sdiv  r1, r1, r3
0x080070c0:  udiv  r1, r1, r3
```

收尾时读到的 `CFSR` 高半字（UFSR）为 0；运行期除数 19.61 只比常量版 17.15 多 2.46 个周期，不像软件例程的量级。

**这件事不能凭内核型号推断。** 同一套 `arm-none-eabi-gcc` 13.2.1 对 `-mcpu=cortex-m3` 就发 `sdiv`
（`-march` 自动取 `armv7-m`），也没有 `-mno-div` 一类开关；连 libgcc 里给这个 multilib 编的
`__udivsi3` 自己也是 `udiv`。「M3 的除法会不会掉进软件例程」取决于手里这颗 M3 有没有除法单元，
既不能写成「Cortex-M3 一律没有硬件除法」，也不能写成「Cortex-M3 都有」——**以反汇编与实测为准**，本次只测了这一块 r1p1。

> [!NOTE]
> 这一小节的三组数：单精度乘加 11 与 100（约 9 倍）、`double` 两侧都是 166、
> `text` 段 31,000 与 31,840（差 840 字节）；决定它们的是同一个选项 `-mfloat-abi`，而它改的是 ABI。
> 中断里的浮点上下文是同一件事的另一面，见《06-更底层/10-中断、并发与内存序.md》第 1.5 小节。

## 6.7 硬件替 CPU 算：CORDIC 与 FMAC

这颗芯片上有两个专门做算术的单元：CORDIC 算三角函数、双曲函数与模／相角，
FMAC 做滤波与向量运算。它们与 FPU 的分工不同：FPU 是内核的一部分，
这两个是挂在总线上的外设，各有自己的时钟门与数据寄存器。

`实测数据`
`Text`

```text
AHB1ENR 使能前 = 0x00000100
AHB1ENR 使能后 = 0x00000118  （bit3 CORDICEN = 0x8、bit4 FMACEN = 0x10）
```

`RCC->AHB1ENR` 里 `0x100` 是 HSI 那一位；两个加速器的时钟门在复位后是关着的，
用之前先置位并读回确认。

**CORDIC 的输入刻度**。要让它算正弦，先得知道输入怎么表示。
这个刻度不是查手册抄来的，是扫出来的：喂一串输入、把正弦与余弦读回来，
再用 `atan2(正弦, 余弦)` 反推硬件实际使用的角度。

`实测数据`
`Text`

```text
     x            SINE(x)     正弦/2^31    COSINE(x)    余弦/2^31   atan2/pi
     0x08000000     418952960    0.195090    2106219008    0.980785   0.062500
     0x20000000    1518499328    0.707106    1518500096    0.707107   0.250000
     0x80000000         -1536   -0.000001   -2147483136   -1.000000  -1.000000
```

最后一列与 `x / 2^31` 每一行都对得上：`0x08000000` 是 0.0625、`0x20000000` 是 0.25、
`0x80000000` 是 −1.0，**因此 π ↔ 2^31，`0x40000000` 就是 π/2**。
标定值也对得上：`SINE(0x40000000)` = 2147483392，除 2^31 得 0.99999995，就是 1.0；
同一处的余弦是 −9472，即 −4.4e−6，是 π/2 处的零残差。

周期数与误差：

`实测数据`

| 算法 | 每次调用周期数 | 最大绝对误差 |
|---|---:|---:|
| CORDIC，精度 4 | **25.02** | 2.889e−05 |
| CORDIC，精度 15 | **33.01** | 1.353e−06 |
| `sinf`（单精度库函数） | **65.04** | — |
| `sin`（双精度库函数） | **1654.54** | — |

条件与第 6.6 小节相同：STM32G431、HSI 16 MHz、`-O2`、同一套 `arm-none-eabi-gcc` 13.2.1。
误差是在 1024 个角度上取的最大绝对值，那些角度落在 `0x10000000` 到 `0x10000FD8`，
也就是 π/8 附近的一小段，**不代表全量程的误差**。

精度 4 与 15 只差 8 个周期，误差却相差 20 倍；两者都比 `sinf` 便宜。
双精度那条贵 25 倍以上，与第 6.6 小节「M4F 没有双精度硬件」是同一件事。

**寄存器字段的一个坑**：`CSR` 里的 `NARGS` 与 `NRES` 不是「参数个数」与「结果个数」，
而是「32 位写的个数」与「32 位读的个数」的 n−1 编码：

`文档`
`Text`

```text
#define CORDIC_NBWRITE_1           (0x00000000U)   /* 一次 32 位写 */
#define CORDIC_NBWRITE_2           CORDIC_CSR_NARGS
#define CORDIC_NBREAD_1            (0x00000000U)   /* 一次 32 位读 */
#define CORDIC_NBREAD_2            CORDIC_CSR_NRES
```

—— `stm32g4xx_hal_cordic.h` 第 266 到 290 行。

正弦与余弦都只吃一个参数、只出一个结果，因此这两位要留 0。
按字面写成 1 时，硬件会去等第二次写、第二次读，`RRDY` 永远不置位——
**实测卡死在轮询 `RRDY` 的那个循环里**，PC 停在 `0x0800969c`，
反汇编出来就是 `ldr r2,[r3,#0]; cmp r2,#0; bge` 三条指令。

`C`

```c
/* tests.c（节选）    CORDIC 的配置与一次运算，NARGS 与 NRES 留 0 */
static void cordic_cfg(uint32_t func, uint32_t precision_cycles)
{
    CORDIC->CSR = func
                | (precision_cycles << CORDIC_CSR_PRECISION_Pos)
                | (0u << CORDIC_CSR_NARGS_Pos)    /* 1 个 32 位参数 */
                | (0u << CORDIC_CSR_NRES_Pos)     /* 1 个 32 位结果 */
                | (0u << CORDIC_CSR_ARGSIZE_Pos)  /* 输入 32 位（Q1.31） */
                | (0u << CORDIC_CSR_RESSIZE_Pos); /* 输出 32 位（Q1.31） */
    (void)CORDIC->RDATA;                          /* 清掉上一次的 RRDY */
}

static int32_t cordic_run(int32_t arg)
{
    CORDIC->WDATA = (uint32_t)arg;
    for (uint32_t i = 0; i < 1000000u; i++) {
        if (CORDIC->CSR & CORDIC_CSR_RRDY) {
            return (int32_t)CORDIC->RDATA;
        }
    }
    cordic_timeout = 1;                           /* 不挂死，让后面的实验还能跑 */
    return 0;
}
```

`CSR` 的复位值实测是 `0x00000050`：`FUNC` 字段 0（余弦）、`PRECISION` 字段 5。

**一处没有定论的现象**（存疑）：最早的版本里，使能 `CORDICEN` 之后不插任何别的访问，
直接配置、直接算，**第一次运算读到 1024**，即 5e−7，与应当得到的 1.0 差得远；
重跑同一个镜像仍然是 1024。后来在配置前后各加了一次 `CSR` 回读，就再没有复现过，
第一次与第二次都是 2147483392。回读证明 `CSR` 的写入是生效的（读到 `0x000000f1`），
因此问题在第一次运算、不在配置；但差几次寄存器访问的时间结果就变了，
**给不出确定结论**。稳妥做法是使能 CORDIC 时钟之后先做一次丢弃的运算。

**FMAC**：8 抽头 FIR，256 个样本，249 个输出。同一组系数、同一批输入，
只改喂数据的节奏，结果差出三个数量级：

`实测数据`
`Text`

```text
装 8 个系数：START 已落  PARAM = 0x02000008  SR = 0x00000001
X1BUFCFG = 0x01008008  X2BUFCFG = 0x00000800  YBUFCFG = 0x01004088
A 逐写逐读（Q = 0）：9009396 周期，读到 248 个输出
C 循环：18197 周期，249 个输出
B 成块搬（Q = 0）：6633 周期，读到 249 个输出，来回 2 轮
C 成块搬（Q = 8）：6633 周期，读到 249 个输出，来回 2 轮
每个输出样本：A 36328 周期，B 26 周期，C 26 周期，C 循环 73 周期
```

`实测数据`

| 做法 | 每次输出周期数 | 说明 |
|---|---:|---|
| A 写一个样本就读一次 | **36,328** | 比 C 循环慢约 500 倍 |
| B 先写满 X1、再把 Y 读空 | **26** | 比 C 循环快约 2.8 倍 |
| C 循环（8 次乘加／输出） | **73** | 基准 |

这一版的缓冲区是 X1 = 128 字、X2 = 8 字、Y = 64 字，空／满水位都放在 2^5 = 32，
`X1_BASE = 8`、`X2_BASE = 0`、`Y_BASE = 136`。`Q` 字段对 FIR 没有影响：
`Q = 0` 与 `Q = 8` 的周期数完全一样，都是 6633，与 RM0440 里那句 `Q = "Don't care"` 对得上。

`文档`

> "The buffer must be configured as follows: X1_BUF_SIZE = N + d1; X2_BUF_SIZE = N; Y_BUF_SIZE = d2"
> "The filter is started by writing to the FMAC_PARAM register with the following bitfield values:
> FUNC = 8 (FIR filter); P = N (number of coefficients); Q = "Don't care"; R = Gain; START = 1"
>
> —— RM0440 §18.3.7 与 §18.3.8

两种节奏的代码差别只有几行（下面是节选，省掉了输出缓冲的边界判断）：

`C`

```c
/* fmac.c（节选）    同一组系数、同一批输入，只改写与读的顺序 */
/* 方式 A：写一个样本，就把 Y 里能读的全读走 —— 9009396 周期 */
for (uint32_t i = 0; i < N_SAMPLES; i++) {
    while ((FMAC->SR & FMAC_SR_YEMPTY) == 0u) { out_fmac[nread++] = (int16_t)FMAC->RDATA; }
    uint32_t guard = 0;
    while (FMAC->SR & FMAC_SR_X1FULL) {   /* X1 满了就走不动，只能先把 Y 读空 */
        while ((FMAC->SR & FMAC_SR_YEMPTY) == 0u) { out_fmac[nread++] = (int16_t)FMAC->RDATA; }
        if (++guard > 1000000u) { fmac_stall = 1; break; }
    }
    FMAC->WDATA = (uint16_t)xp[N_TAPS + i];
}

/* 方式 B：先把 X1 写到满，再把 Y 读空，来回倒 —— 6633 周期 */
for (uint32_t guard = 0; guard < 2000000u; guard++) {
    while ((written < N_SAMPLES) && ((FMAC->SR & FMAC_SR_X1FULL) == 0u)) {
        FMAC->WDATA = (uint16_t)xp[N_TAPS + written];
        written++;
    }
    while ((FMAC->SR & FMAC_SR_YEMPTY) == 0u) { out_fmac[nread++] = (int16_t)FMAC->RDATA; }
    if ((written >= N_SAMPLES) && (FMAC->SR & FMAC_SR_YEMPTY)) { break; }
}
```

> [!IMPORTANT]
> **怎么喂数据比用不用硬件重要得多。**
> 同一个外设、同一组系数、同一批数据，逐写逐读是 36,328 周期一个输出，
> 成块搬是 26 周期：后者还比纯 C 循环快 2.8 倍，前者慢 500 倍。
> 硬件算得快不快，取决于有没有让它连续地算。

A 与 B 各自复现过两次，两次的 A 都在 9.00e6 周期上下。
机制没有单独验证过，推测是 X1 一空单元就停下来等、每次重启的代价高，
**这一条存疑**。

**数值对照里另有一处存疑**。第一次跑的是小缓冲区（X1 = 16、Y = 8、水位都是 8），
同样是 A 那种逐写逐读，与 C 循环逐点比：

`实测数据`
`Text`

```text
装 8 个系数：START 已落，30 周期  PARAM = 0x02000008  SR = 0x00000001
FMAC：9009571 周期，读到 242 个输出  SR = 0x00000001
C 循环：18192 周期
前 4 个输出（FMAC 的第 i 个对应 C 的第 i+7 个）：
   FMAC[0] =  -9120   C[7] =  -9120
   FMAC[1] =  -8800   C[8] =  -8800
   FMAC[2] =  -8480   C[9] =  -8480
   FMAC[3] =  -8160   C[10] = -8160
与 C 循环逐点比：0 个点不同（共 249 个），最大绝对差 0
```

**换成大缓冲区（128/64）之后就不同了**：248 个点全部不同（实际比了 248 个），
最大绝对差 2880，前四个输出每个都差 192。
`CR` 的 `RESET` 位清不掉 X1 里残留的样本，
改用 `RCC->AHB1RSTR` 的 `FMAC_AHB1RSTR_FMACRST` 做整块复位，仍然不同——
**原因没有查清，这一条存疑**。
要用这组结论时，以「小缓冲区那一版逐点相同」为准，并在换缓冲区配置之后重新对一遍数值。

> [!NOTE]
> CORDIC 把三角函数从 `sinf` 的 65 周期压到 25 到 33 周期，
> 代价是输入要做成 π ↔ 2^31 的定点刻度；
> FMAC 的 26 与 36,328 说明加速器的收益取决于数据怎么喂，
> 而不取决于它有没有被用上。

# 第 7 节 速查表

| 常用件 | 一句话用途 | 典型坑 |
|---|---|---|
| `sizeof` | 看对象里到底有几个字节 | 加一个 `virtual` 就跳一档；基类子对象不占位（空基类优化） |

| `static_cast` | 编译期算好偏移的转换 | 交叉转换直接报错，要用 `dynamic_cast` |
| `reinterpret_cast` | 按位重新解释 | **多继承时做向上转换会指错位置**，且不报错 |
| `dynamic_cast` | 运行期查类型信息 | 一次约十次虚调用的时间；MCU 上要慎用 |
| `typeid(x).name()` | 拿到类型名字符串 | 是修饰名，格式由实现决定，不能用来判断类型 |
| `-fno-rtti` | 砍掉 `type_info` 与查表代码 | 本例省 1332 字节；用 `typeid`／`dynamic_cast` 的地方会编译失败 |
| `-fno-exceptions` | 关掉异常 | **在 ARM 裸机上一点体积也省不下来** |
| `__dso_handle`、`_fini` | `-nostartfiles` 下全局对象析构注册要用的两个符号 | 删了它们链接会报 `undefined reference` |
| `--gc-sections` | 让链接器丢掉没引用的代码与数据 | 不加它时体积数字里混着整块没用的库代码 |
| `FPU`（Cortex-M4F） | 单精度乘加从 100 周期降到 11 周期 | `double` 没有硬件，仍走软件例程；不开 `SCB->CPACR` 的 CP10/CP11 会进 `UsageFault` |
| `CORDIC` / `FMAC` | 硬件算三角函数与 FIR 滤波 | 收益取决于数据怎么喂；`CORDIC` 的 `NARGS`／`NRES` 是 n−1 编码，按字面写成 1 会让 `RRDY` 永不置位 |

## A.4 `vbase1.cpp`

第 4.1 小节引用的那几行由下面这份程序打出：

`C++`

```cpp
/* vbase1.cpp   编译：g++ -std=c++17 vbase1.cpp -o vbase1
   只有一个虚基类时的偏移槽取值 */
#include <cstdio>
#include <cstdint>
#include <cstddef>

class A { public: virtual void a() { } int x = 1; };
class V : virtual public A { public: virtual void v() { } int y = 2; };
class W : public V { public: virtual void w() { } int z = 3; };

using Slot = void (*)();
using Vptr = Slot *;

static void show(const char *name, const void *sub, const void *obj) {
    const Vptr vp = *reinterpret_cast<const Vptr *>(sub);
    std::printf("%-12s 偏移 %-4td  vptr[-3]= %-5td  vptr[-2]= %td\n", name,
                reinterpret_cast<const char *>(sub) - reinterpret_cast<const char *>(obj),
                static_cast<std::ptrdiff_t>(reinterpret_cast<const std::intptr_t *>(vp)[-3]),
                static_cast<std::ptrdiff_t>(reinterpret_cast<const std::intptr_t *>(vp)[-2]));
}

int main() {
    V v; W w;
    std::printf("sizeof(V)=%zu sizeof(W)=%zu\n", sizeof(V), sizeof(W));
    std::printf("V 里 A 的偏移 = %td\n",
                reinterpret_cast<const char *>(static_cast<A *>(&v)) - reinterpret_cast<const char *>(&v));
    show("V 主", &v, &v);
    show("V 里 A", static_cast<A *>(&v), &v);
    std::printf("W 里 A 的偏移 = %td\n",
                reinterpret_cast<const char *>(static_cast<A *>(&w)) - reinterpret_cast<const char *>(&w));
    show("W 主", &w, &w);
    show("W 里 V", static_cast<V *>(&w), &w);
    show("W 里 A", static_cast<A *>(&w), &w);
    return 0;
}
```

`实测数据`
`Text`

```text
sizeof(V)=32 sizeof(W)=32
V 里 A 的偏移 = 16
V 主        偏移 0     vptr[-3]= 16     vptr[-2]= 0
V 里 A      偏移 16    vptr[-3]= 0      vptr[-2]= -16
W 里 A 的偏移 = 16
W 主        偏移 0     vptr[-3]= 16     vptr[-2]= 0
W 里 V      偏移 0     vptr[-3]= 16     vptr[-2]= 0
W 里 A      偏移 16    vptr[-3]= 0      vptr[-2]= -16
```

`W` 继承 `V`，虚基类还是同一个 `A`，偏移槽仍然是 16——
**偏移是从当前虚指针的地址点算到虚基类的距离，不是从对象开头算的。**
这条规则在第 4.2 小节的四行数据上同样成立。

## A.5 `semihosting.h`

Cortex-M 侧的几份程序都包含它。放在 `06-更底层/README.md` 第五节说明的那条输出通道上：

`C`

```c
/* semihosting.h    Cortex-M 侧的输出通道：bkpt 0xAB + SYS_WRITE0，QEMU 直接接到 stdout。
   主机上退化为 stdout，因此同一份源码在主机上也编得过。
   编译（Cortex-M）：arm-none-eabi-g++ -mcpu=cortex-m3 -mthumb -nostartfiles
                     --specs=nosys.specs -T STM32F103XX_FLASH.ld startup_stm32f103xe.s xx.cpp -o xx.elf */
#ifndef SEMIHOSTING_H
#define SEMIHOSTING_H

#include <stdint.h>

#if defined(__arm__)
/* r0 放功能号、r1 放参数，bkpt 0xAB 陷入调试器；三条指令写在同一个 asm 块里，
   不依赖编译器的寄存器分配（只写 r0/r1 的 register 变量在 -O2 下会被优化坏）。 */
static void sh_puts(const char *s)
{
    __asm__ volatile ("mov r1, %0\n\t"
                      "movs r0, #4\n\t"    /* 0x04 = SYS_WRITE0 */
                      "bkpt 0xAB"
                      : : "r"(s) : "r0", "r1", "memory");
}
#else
#include <stdio.h>
static void sh_puts(const char *s) { fputs(s, stdout); }
#endif

static void sh_dec32(uint32_t v)
{
    char buf[11];
    int i = 10;
    buf[i] = '\0';
    do { buf[--i] = (char)('0' + (v % 10u)); v /= 10u; } while (v != 0u);
    sh_puts(&buf[i]);
}

/* 打一行「标签 = 十进制值」 */
static void sh_row(const char *label, uint32_t v)
{
    sh_puts(label);
    sh_dec32(v);
    sh_puts("\r\n");
}
#endif
```
